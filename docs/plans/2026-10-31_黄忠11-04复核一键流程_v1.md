### 4.2 唯一 commit 覆盖率（补充口径，防「同 HEAD 重跑注水」）

```powershell
# 只读核算：按 commit.sha 去重 PASS 单 / merge_done（演练单不计；与 go-no-go 工具口径一致）
cd C:\项目\老田的AI助手
C:\项目\py312\python.exe -X utf8 -c @"
import json, glob
shas = set()
for p in glob.glob('work/huangzhong/acceptance_*.json'):
    try:
        d = json.load(open(p, encoding='utf-8'))
    except Exception:
        continue
    if str(d.get('verdict') or '').upper() != 'PASS':
        continue
    sha = ((d.get('commit') or {}).get('sha') or '').strip()
    if sha:
        shas.add(sha)
print('unique_pass_commits =', len(shas))
"@
```

```text

└─ ②/③/⑥ 缺 ──→ 补演练/复核清单/抽查记录后重跑
```
