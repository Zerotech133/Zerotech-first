```powershell

# 输出 version 与 artifact 目录

# 2) 校验 artifact 完整性

# 3) 校验生产当前服务内容（切换前会 mismatch，用于证明现状）
python tools/release_legion_ide.py --verify-http <artifact_dir> --base http://127.0.0.1:8787/extensions

# 4) 提升指针（仅写 current.json，不重启）
python tools/release_legion_ide.py --promote <version>

# 5) 回滚指针
python tools/release_legion_ide.py --rollback-to <version>
```
