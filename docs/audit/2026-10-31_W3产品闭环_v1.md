w['prior_vol'] = w['vol'].rolling(20).mean().shift(1)
w['signal'] = (w['close'] > w['prior_high']) & (w['vol'] > 1.5 * w['prior_vol'])
oos = (w['trade_date'] >= '20250701') & (w['trade_date'] <= '20261016')
