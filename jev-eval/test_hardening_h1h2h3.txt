# test_hardening_h1h2h3.txt - H1/H2/H3 加固部署回归测试（OPS-1001e，2026-10-04）
# 运行（独立进程，判例：含 exit 的脚本不可在交互会话内 dot-source）：
#   powershell -NoProfile -Command ". ([ScriptBlock]::Create([IO.File]::ReadAllText('<本文件>', [Text.Encoding]::UTF8)))"
# 约定：exit 0 = 全过；exit N = FAIL 数。断言目标 = 防加固回退（Windows 更新/误操作改任一设置即红）。
# 落盘 .txt 非 .ps1：全盘 .ps1 静默秒删判例（OPS-1001d）。
$ErrorActionPreference = 'Continue'
$fail = 0; $n = 0
function Check($name, $cond) {
    $script:n++
    if ($cond) { Write-Output ('[PASS] ' + $name) }
    else { $script:fail++; Write-Output ('[FAIL] ' + $name) }
}

# ---- T1 N1Check v1.1 源码静态断言（注册表 base64 解码） ----
$regSrc = (Get-ItemProperty HKLM:\SOFTWARE\AIStation\N1Check -ErrorAction SilentlyContinue).source
Check 'T1-1 注册表 source 存在' ([bool]$regSrc)
$dec = $null
if ($regSrc) { try { $dec = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($regSrc)) } catch {} }
Check 'T1-2 base64 合法可解码（roundtrip 完整性）' ($null -ne $dec -and $dec.Length -gt 5000)
if ($dec) {
    Check 'T1-3 头注释 v1.1 每日化标识' ($dec -match 'N1Check v1\.1' -and $dec -match 'daily 07:30')
    Check 'T1-4 $day 动态取当日' ($dec -match '\$day = \(Get-Date\)\.ToString\(''yyyy-MM-dd''\)')
    Check 'T1-5 窗口守卫行已彻底移除' ($dec -notmatch '2026-10-02 05:00')
    Check 'T1-6 one-shot 硬编码日期已移除' ($dec -notmatch '\$day = ''2026-10-02''')
    Check 'T1-7 FAIL→flag 告警路径保留' ($dec -match 'N1_CHECK_FAIL_')
    Check 'T1-8 报告落盘模板保留' ($dec -match 'n1_report_')
    Check 'T1-12 晨检含 F2 密钥守卫检查项（BACKUP_KEY_LOST）' ($dec -match 'BACKUP_KEY_LOST')
    Check 'T1-13 晨检含探活检查项（ENGINE_DOWN）' ($dec -match 'ENGINE_DOWN')
    Check 'T1-14 晨检含异地检查项（OneDrive+backups_mirror）' ($dec -match 'OneDrive' -and $dec -match 'backups_mirror')
}

# ---- T1b 加固家族脚本载体完整性（增量：OffsiteSync / BackupDeploy 分段源码从未入测） ----
$osSrc = (Get-ItemProperty HKLM:\SOFTWARE\AIStation\OffsiteSync -ErrorAction SilentlyContinue).source
Check 'T1b-1 异地任务注册表源码存在（H1 载体完整性）' ([bool]$osSrc)
$osDec = $null
if ($osSrc) { try { $osDec = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($osSrc)) } catch {} }
Check 'T1b-2 OffsiteSync base64 可解码' ($null -ne $osDec -and $osDec.Length -gt 3000)
$bdp = Get-ItemProperty HKLM:\SOFTWARE\AIStation\BackupDeploy -ErrorAction SilentlyContinue
$bdDec = $null
if ($bdp.part1) { try { $bdDec = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($bdp.part1 + $bdp.part2)) } catch {} }
Check 'T1b-3 BackupDeploy part1+part2 拼接可解码（分段判例首次入测）' ($null -ne $bdDec -and $bdDec.Length -gt 8000)

# ---- T2 H1/H2 任务设置断言 ----
$bk = Get-ScheduledTask -TaskName 'AI工作站每日备份' -ErrorAction SilentlyContinue
$os = Get-ScheduledTask -TaskName 'AI工作站异地同步' -ErrorAction SilentlyContinue
Check 'T2-1 每日备份任务存在' ([bool]$bk)
Check 'T2-2 异地同步任务存在' ([bool]$os)
if ($bk) {
    Check 'T2-3 H1: 备份 StartWhenAvailable=True（错过后补跑）' ($bk.Settings.StartWhenAvailable -eq $true)
    Check 'T2-4 H2: 备份 WakeToRun=True（凌晨唤醒）' ($bk.Settings.WakeToRun -eq $true)
    Check 'T2-5 备份仍为 SYSTEM 服务账户（部署基线未动）' ($bk.Principal.UserId -eq 'SYSTEM' -and $bk.Principal.LogonType -eq 'ServiceAccount')
}
if ($os) {
    Check 'T2-6 H1: 异地 StartWhenAvailable=True（错过后补跑）' ($os.Settings.StartWhenAvailable -eq $true)
    Check 'T2-7 异地 WakeToRun=False（设计位：04:30 时已醒，防误改）' ($os.Settings.WakeToRun -eq $false)
}
if ($bk) { Check 'T2-8 备份 trigger 时刻仍为 03:00（定时基线防误改）' (@($bk.Triggers)[0].StartBoundary -match 'T03:00') }
if ($os) { Check 'T2-9 异地 trigger 时刻仍为 04:30（定时基线防误改）' (@($os.Triggers)[0].StartBoundary -match 'T04:30') }
$wd = Get-ScheduledTask -TaskName 'AI工作站引擎探活' -ErrorAction SilentlyContinue
Check 'T2-10 引擎探活任务存在且 Ready（加固家族旁证组件）' ([bool]$wd -and $wd.State -eq 'Ready')

# ---- T3 H2 电源计划断言（0x 索引解析，避开中文标签编码依赖） ----
$pq = (powercfg /query SCHEME_CURRENT 238c9fa8-0aad-41ed-83f4-97be242c8f20 bd3b718a-0680-4d9d-8ab2-e1d2b4ac806d 2>$null) -join "`n"
Check 'T3-1 唤醒计时器设置项查询有输出' ($pq.Length -gt 0)
$idx = @([regex]::Matches($pq, '0x0*([0-9a-fA-F]+)') | ForEach-Object { $_.Groups[1].Value })
Check 'T3-2 唤醒计时器 AC=启用(1)' ($idx.Count -ge 2 -and $idx[0] -eq '1')
Check 'T3-3 唤醒计时器 DC=启用(1)' ($idx.Count -ge 2 -and $idx[1] -eq '1')

# ---- T4 H3 晨检任务 trigger 断言 ----
$n1 = Get-ScheduledTask -TaskName 'AI工作站N1晨检' -ErrorAction SilentlyContinue
Check 'T4-1 N1 晨检任务存在且 Ready' ([bool]$n1 -and $n1.State -eq 'Ready')
if ($n1) {
    $tr = @($n1.Triggers)[0]
    Check 'T4-2 trigger 为每日类型' ($tr.CimClass.CimClassName -match 'Daily')
    Check 'T4-3 trigger 时刻 07:30' ($tr.StartBoundary -match 'T07:30')
    Check 'T4-4 DaysInterval=1（每天，防退化为 one-shot/每周）' ($tr.DaysInterval -eq 1)
    $n1i = $n1 | Get-ScheduledTaskInfo
    Check 'T4-5 NextRunTime 已排期在未来（one-shot 特征 Next 为空已消除）' ($null -ne $n1i.NextRunTime -and $n1i.NextRunTime -gt (Get-Date))
    Check 'T4-6 N1 Action 仍走注册表解码模式（Arguments 引用 N1Check，防漂移到磁盘路径）' ((@($n1.Actions)[0].Arguments -join ' ') -match 'N1Check')
}
if ($bk) { Check 'T4-7 备份 Action 引用 BackupDeploy（注册表模式未漂移）' ((@($bk.Actions)[0].Arguments -join ' ') -match 'BackupDeploy') }
if ($os) { Check 'T4-8 异地 Action 引用 OffsiteSync（注册表模式未漂移）' ((@($os.Actions)[0].Arguments -join ' ') -match 'OffsiteSync') }

# ---- T6 留档一致性（aios_deploy 与注册表零漂移） ----
$arch = 'C:\aios_deploy\n1_check_v1_1.txt'
Check 'T6-1 留档文件存在' (Test-Path $arch)
if ((Test-Path $arch) -and $dec) {
    $archTxt = [IO.File]::ReadAllText($arch, [Text.Encoding]::UTF8)
    Check 'T6-2 留档内容与注册表解码源码一致' ($archTxt -eq $dec)
}

# ---- T7 H3 行为级测试（增量：FAIL→flag 告警路径自部署以来从未执行过——最后一公里缺口） ----
# monkeypatch 模式（python 惯例的 PS 等价）：源码路径文本替换到 temp + 前置 mock 函数 + 独立子进程执行
# （v1.1 的 Fail() 用 $script: 前缀，同进程执行会污染 harness 计数——必须子进程隔离）
Check 'T7-0 前置：v1.1 源码已解码可用' ([bool]$dec)
if ($dec) {
    $tRoot = Join-Path $env:TEMP ('n1t' + (Get-Date -Format 'HHmmss'))
    $mock = "function Get-ScheduledTask { param(`$Name) [pscustomobject]@{ Actions = @([pscustomobject]@{ Arguments = 'OffsiteSync-mock' }) } }`r`nfunction Get-ScheduledTaskInfo { param(`$InputObject) [pscustomobject]@{ LastTaskResult = 0; LastRunTime = [datetime]::Today.AddHours(4.5) } }"
    function Invoke-N1Case($tag, $happy) {
        $r = Join-Path $tRoot $tag
        $fakeE = Join-Path $r 'fakeE'; $proj = Join-Path $fakeE 'fakeproj'
        $mirror = Join-Path $r 'mirror'; $od = Join-Path $r 'od'
        $alertsT = Join-Path $proj 'jev-eval\alerts'
        $day2 = (Get-Date).ToString('yyyy-MM-dd')
        $bkT = Join-Path $proj 'backups'
        New-Item -ItemType Directory -Force -Path $alertsT, (Join-Path $proj 'jev-eval'), $od | Out-Null
        Set-Content (Join-Path $proj 'jev-eval\backup_alert.py') '# sentinel'
        if ($happy) {
            $setDir = Join-Path $mirror ($day2 + '_0300')
            New-Item -ItemType Directory -Force -Path $setDir, (Join-Path $bkT ($day2 + '_0300')), (Join-Path $od ($day2 + '_0300')), (Join-Path $od '2020-01-01_0000') | Out-Null
            1..13 | ForEach-Object { Set-Content (Join-Path $setDir ('f' + $_ + '.bin')) 'x'; Set-Content (Join-Path $od ($day2 + '_0300\f' + $_ + '.bin')) 'x' }
            "[0305] OK   quant_code.tar.gz files=1234`n[0305] DONE failed=0" | Set-Content (Join-Path $bkT ($day2 + '_0300\backup.log'))
            '[0430] OK set=' + $day2 + ' files=13' | Set-Content (Join-Path $r 'offsite.log')
        }
        $patched = $dec.Replace("'E:\'", ("'" + $fakeE + "'"))
        $patched = $patched.Replace("'D:\backups_mirror'", ("'" + $mirror + "'"))
        $patched = $patched.Replace("'C:\Temp\offsite_sync.log'", ("'" + (Join-Path $r 'offsite.log') + "'"))
        $patched = $patched.Replace("'C:\Users\Administrator\OneDrive\backups_mirror'", ("'" + $od + "'"))
        $caseFile = Join-Path $r 'case.txt'
        [IO.File]::WriteAllText($caseFile, $mock + "`r`n" + $patched, [Text.Encoding]::UTF8)
        $runner = ". ([ScriptBlock]::Create([IO.File]::ReadAllText('" + $caseFile + "', [Text.Encoding]::UTF8)))"
        powershell -NoProfile -Command $runner | Out-Null
        return @{ r = $r; proj = $proj; alerts = $alertsT }
    }
    # -- 用例 B：happy 路径（全好数据 → ALL PASS / 无 flag） --
    $b = Invoke-N1Case 'happy' $true
    $bRep = @(Get-Content (Join-Path $b.proj ('jev-eval\n1_report_' + (Get-Date).ToString('yyyy-MM-dd') + '.txt')) -ErrorAction SilentlyContinue)
    $bFlag = @(Get-ChildItem $b.alerts -Filter 'N1_CHECK_FAIL_*.flag' -ErrorAction SilentlyContinue)
    Check 'T7-B1 happy: 报告生成且尾行 ALL PASS' ($bRep.Count -gt 0 -and $bRep[-1] -match 'ALL PASS')
    Check 'T7-B2 happy: 十项检查全 [OK]' (@($bRep | Where-Object { $_ -match '^\[OK\]' }).Count -eq 10)
    Check 'T7-B3 happy: 无 FAIL flag' ($bFlag.Count -eq 0)
    # -- 用例 A：FAIL 路径（无当日集/无 offsite log/OneDrive 缺集 → flag 必须生成） --
    $a = Invoke-N1Case 'fail' $false
    $aRep = @(Get-Content (Join-Path $a.proj ('jev-eval\n1_report_' + (Get-Date).ToString('yyyy-MM-dd') + '.txt')) -ErrorAction SilentlyContinue)
    $aFlag = @(Get-ChildItem $a.alerts -Filter 'N1_CHECK_FAIL_*.flag' -ErrorAction SilentlyContinue)
    Check 'T7-A1 FAIL: 报告尾行 FAILURES=N 且 N>=3（缺集/缺log/缺OneDrive集）' ($aRep.Count -gt 0 -and $aRep[-1] -match 'FAILURES=(\d+)' -and [int]$Matches[1] -ge 3)
    Check 'T7-A2 FAIL: 缺集首因正确报告' (@($aRep | Where-Object { $_ -match 'no backup set' }).Count -eq 1)
    Check 'T7-A3 FAIL: N1_CHECK_FAIL flag 恰好生成 1 个' ($aFlag.Count -eq 1)
    Check 'T7-A4 FAIL: flag 内容含 [FAIL] 报告行' ($aFlag.Count -eq 1 -and ((Get-Content $aFlag[0].FullName -Raw) -match '\[FAIL\]'))
    Remove-Item $tRoot -Recurse -Force -ErrorAction SilentlyContinue
}

# ---- 汇总 ----
Write-Output ('=' * 44)
Write-Output ('TOTAL=' + $n + ' FAIL=' + $fail)
exit $fail
